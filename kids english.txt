    // Web Audio Synthesizer for Fun Sound Effects
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    function playTone(freq, type, duration) {
      if (audioCtx.state === 'suspended') audioCtx.resume();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = type;
      osc.frequency.value = freq;
      gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + duration);
    }

    function playSuccessChime() {
      playTone(523.25, 'sine', 0.15); // C5
      setTimeout(() => playTone(659.25, 'sine', 0.15), 100); // E5
      setTimeout(() => playTone(783.99, 'sine', 0.3), 200); // G5
    }

    function playPopSound() {
      playTone(300, 'triangle', 0.08);
    }

    /* Initializing Speech Engine & Voices */
    function populateVoiceList() {
      if (!synth) return;
      let voices = synth.getVoices();
      let picker = document.getElementById('voice-picker');
      picker.innerHTML = '';

      // Filter standard English voices
      let englishVoices = voices.filter(v => v.lang.startsWith('en'));

      if (englishVoices.length === 0) englishVoices = voices;

      englishVoices.forEach((v, index) => {
        let option = document.createElement('option');
        option.textContent = `${v.name} (${v.lang})`;
        option.value = index;
        picker.appendChild(option);
      });

      // Default to preferred English voice
      let defaultVoice = englishVoices.find(v => v.lang === 'en-US' || v.lang.startsWith('en')) || englishVoices[0];
      if (defaultVoice) {
        selectedVoice = defaultVoice;
        picker.value = englishVoices.indexOf(defaultVoice);
      }
    }

    if (speechSynthesis.onvoiceschanged !== undefined) {
      speechSynthesis.onvoiceschanged = populateVoiceList;
    }

    function changeVoice() {
      let voices = synth.getVoices().filter(v => v.lang.startsWith('en'));
      let selectedIndex = document.getElementById('voice-picker').value;
      if (voices[selectedIndex]) {
        selectedVoice = voices[selectedIndex];
        speakText("Hi! I am your native English teacher!");
      }
    }

    function speakText(text, rate = 0.85) {
      if (!synth) return;
      synth.cancel(); // Stop ongoing speech
      let utterance = new SpeechSynthesisUtterance(text);
      if (selectedVoice) utterance.voice = selectedVoice;
      utterance.rate = rate; // Slightly slower for kids
      utterance.pitch = 1.1; // Cheerful tone for kids
      synth.speak(utterance);
    }

    }

    function initSpeechRecognition() {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRecognition) {
        document.getElementById('speech-feedback').innerText = "Speech Recognition not supported on this browser version.";
        return;
      }

      recognition = new SpeechRecognition();
      recognition.lang = 'en-US';
      recognition.interimResults = false;

      recognition.onstart = () => {
        isRecording = true;
        document.getElementById('mic-button').classList.add('recording');
        document.getElementById('speech-feedback').innerText = "Listening... Speak now!";
      };

      recognition.onresult = (event) => {
        const transcript = event.results[0][0].transcript.toLowerCase();
        const target = practicePhrases[activePracticeIndex].phrase.toLowerCase();

        document.getElementById('mic-button').classList.remove('recording');
        isRecording = false;

        if (transcript.includes(target) || target.includes(transcript)) {
          document.getElementById('speech-feedback').innerHTML = `🎉 Perfect! You said: <strong>"${event.results[0][0].transcript}"</strong>`;
          addStars(3);
          speakText("Awesome pronunciation!");
        } else {
          document.getElementById('speech-feedback').innerHTML = `👍 Good try! Heard: "${event.results[0][0].transcript}". Try again!`;
          speakText("Good try! Let's say it together.");
        }
      };

      recognition.onerror = () => {
        document.getElementById('mic-button').classList.remove('recording');
        isRecording = false;
        document.getElementById('speech-feedback').innerText = "Could not hear you clearly. Tap mic to try again!";
      };

      recognition.onend = () => {
        document.getElementById('mic-button').classList.remove('recording');
        isRecording = false;
      };
    }

    function toggleSpeechRecognition() {
      if (!recognition) initSpeechRecognition();
      if (!recognition) return;

      if (isRecording) {
        recognition.stop();
      } else {
        recognition.start();
      }
    }

       VIEW 4: MINI QUIZ & STICKERS
       ========================================== */
    function setupNewQuiz() {
      const randomItem = vocabData[Math.floor(Math.random() * vocabData.length)];
      currentQuizAnswer = randomItem;

      // Pick 3 options
      let options = [randomItem];
      while (options.length < 4) {
        let rand = vocabData[Math.floor(Math.random() * vocabData.length)];
        if (!options.find(o => o.text === rand.text)) options.push(rand);
      }
      options.sort(() => Math.random() - 0.5);

      const optionsContainer = document.getElementById('quiz-options');
      optionsContainer.innerHTML = options.map(opt => `
        <div class="quiz-card" onclick="checkQuizAnswer('${opt.text}')">
          <div style="font-size:45px;">${opt.emoji}</div>
          <div style="font-size:16px; font-weight:700; margin-top:5px;">${opt.text}</div>
        </div>
      `).join('');

      playQuizAudio();
    }

    function playQuizAudio() {
      if (currentQuizAnswer) {
        speakText(`Find the ${currentQuizAnswer.text}`);
      }
    }

    function checkQuizAnswer(selectedText) {
      if (selectedText === currentQuizAnswer.text) {
        addStars(2);
        speakText("Great job! That's right!");
        setTimeout(setupNewQuiz, 1200);
      } else {
        speakText("Oops! Try again!");
      }
    }

            el.classList.add('unlocked');
          }
        }
      });
